title: Tiramisu Truffles
type: Dessert
parts:
  - name: Tiramisu truffles — source batch yield 9
    ingredients:
      - 100 g Savoiardi biscuits
      - 100 g mascarpone
      - 25 g espresso
      - 5 g cocoa powder
      - 50 g hazelnuts, toasted and roughly chopped
      - 15 g caster sugar
      - Additional cocoa powder, for dusting
    instructions:
      - step: 1
        description: Cut each biscuit into 5 pieces.
      - step: 2
        description: Put the biscuit pieces, mascarpone, espresso, 5 g cocoa powder, hazelnuts, and caster sugar in a mixing bowl.
      - step: 3
        description: Combine and leave to rest for a few minutes, until the biscuits have absorbed some moisture.
      - step: 4
        description: Mix thoroughly, shape into 30 g balls, and dust lightly with cocoa powder.
  - name: Sweet mascarpone — source batch yield 550 g (unverified)
    ingredients:
      - 250 g mascarpone
      - 50 g double cream
      - 40 g icing sugar
      - Seeds from 1/2 vanilla pod
    instructions:
      - step: 1
        description: Whisk all the ingredients together.
      - step: 2
        description: Transfer to a piping bag.
  - name: Plating — one portion
    ingredients:
      - 3 prepared tiramisu truffles
      - 60 g caramel sauce
      - 45 g sweet mascarpone
      - 15 ml hazelnuts, toasted and roughly chopped (source volume measure)
      - Cocoa powder, for dusting
      - Icing sugar, for dusting
    instructions:
      - step: 1
        description: Re-dust the truffles with cocoa powder.
      - step: 2
        description: Drizzle the caramel sauce in a continuous zig-zag on a white starter plate. Pipe 3 evenly spaced dollops of sweet mascarpone.
      - step: 3
        description: Top each dollop with a truffle, dust with cocoa powder and icing sugar, and scatter over the toasted hazelnuts.
